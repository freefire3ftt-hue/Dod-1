local function toHex(c)
    return string.format("#%02X%02X%02X", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
end

local function roundTo(v, d)
    d = d or 0
    if d <= 0 then return math.floor(v + 0.5) end
    local m = 10 ^ d
    return math.floor(v * m + 0.5) / m
end

local function fmtNum(v, d)
    d = d or 0
    if d <= 0 then return tostring(math.floor(v + 0.5)) end
    local s = string.format("%." .. d .. "f", v)
    if s:find("%.") then s = s:gsub("0+$", ""):gsub("%.$", "") end
    return s
end

local ORIGINAL_THEME = derive({ accent = Color3.fromRGB(255, 112, 22), glow = Color3.fromRGB(255, 196, 96), border = Color3.fromRGB(54, 34, 70), textActive = Color3.fromRGB(255, 130, 40), bgMain = Color3.fromRGB(10, 6, 16), bgHeader = Color3.fromRGB(17, 10, 26), bgCard = Color3.fromRGB(19, 11, 29) })

local function makeTheme(base)
    local glow = Color3.new(math.min(1, base.R * 0.5 + 0.5), math.min(1, base.G * 0.5 + 0.5), math.min(1, base.B * 0.5 + 0.5))
    local border = Color3.new(math.clamp(base.R * 0.3 + 0.05, 0, 1), math.clamp(base.G * 0.3 + 0.05, 0, 1), math.clamp(base.B * 0.3 + 0.05, 0, 1))
    local brightness = base.R * 0.299 + base.G * 0.587 + base.B * 0.114
    return derive({
        accent = base, glow = glow, border = border, textActive = brightness > 0.4 and base or glow,
        bgMain = Color3.new(0.03 + base.R * 0.05, 0.03 + base.G * 0.05, 0.04 + base.B * 0.05),
        bgHeader = Color3.new(0.04 + base.R * 0.06, 0.04 + base.G * 0.06, 0.05 + base.B * 0.06),
        bgCard = Color3.new(0.04 + base.R * 0.06, 0.04 + base.G * 0.06, 0.05 + base.B * 0.06),
    })
end

local function colorMatch(c1, c2, tol)
    tol = tol or 0.02
    return math.abs(c1.R - c2.R) < tol and math.abs(c1.G - c2.G) < tol and math.abs(c1.B - c2.B) < tol
end

    Theme.reg(inst, function(c, animate)

        local r = Theme.regs[i]
        if r.inst and not r.inst.Parent then table.remove(Theme.regs, i)
        else pcall(r.fn, newT, true) end
    end
end

    if not part:IsA("Part") then return false end
    if part.Shape ~= Enum.PartType.Ball then return false end
    if part.Size.X < DodgeConfig.SphereMinSize then return false end
    if part.AssemblyLinearVelocity.Magnitude < DodgeConfig.SphereMinSpeed then return false end

end

    if rescan then DodgeUpdateFilter() end
    local parts, seen = {}, {}
    if BallsContainer then collectFrom(BallsContainer, parts, seen) end
    if PracticeContainer then collectFrom(PracticeContainer, parts, seen) end
    if BallsFolder3 then
        for _, d in ipairs(BallsFolder3:GetDescendants()) do
            if d:IsA("BasePart") and not seen[d] and not isPlayerModel(d) then seen[d] = true; tInsert(parts, d) end

    end

        if child ~= BallsContainer and child ~= PracticeContainer and child ~= BallsFolder3 then
            if not Players:GetPlayerFromCharacter(child) then
                if child:IsA("BasePart") and not seen[child] and (matchesKeyword(child.Name) or isSphereBall(child)) then seen[child] = true; tInsert(parts, child)
                elseif child:IsA("Folder") or child:IsA("Model") then
                    if matchesKeyword(child.Name) then collectFrom(child, parts, seen)
                    else for _, sub in ipairs(child:GetChildren()) do if sub:IsA("BasePart") and not seen[sub] and (matchesKeyword(sub.Name) or isSphereBall(sub)) then seen[sub] = true; tInsert(parts, sub) end end end
                end
            end
        end
    end
    for i = #BallCache, 1, -1 do BallCache[i] = nil end
    for i, part in ipairs(parts) do if i > DodgeConfig.MaxBalls then break end; tInsert(BallCache, part) end
    for part in pairs(BallSample) do if not part.Parent then BallSample[part] = nil; BallVelocity[part] = nil end end
end

local function velocityOf(part, pos, now)
    local vel = part.AssemblyLinearVelocity
    if vel.Magnitude > 1 then BallVelocity[part] = vel; BallSample[part] = {pos, now}; return vel end
    local s = BallSample[part]
    if s then local dt = now - s[2]; if dt > 0.05 then local est = (pos - s[1]) / dt; BallVelocity[part] = est; BallSample[part] = {pos, now}; return est end; return BallVelocity[part] or vel end
    BallSample[part] = {pos, now}; return BallVelocity[part] or vel
end

local function boxGap(lp, r)
    local dx = mAbs(lp.X) - (DodgeConfig.BoxWidth * 0.5 + r)
    local dy = mAbs(lp.Y - DodgeConfig.BoxOffsetY) - (DodgeConfig.BoxHeight * 0.5 + r)
    local dz = mAbs(lp.Z) - (DodgeConfig.BoxDepth * 0.5 + r)
    return mMax(dx, mMax(dy, dz))
end

local function clearanceAlong(origin, dir)
    local probe = Workspace:Raycast(origin, dir * DodgeConfig.ClearanceProbe, dodgeRay)
    if not probe then return DodgeConfig.ClearanceProbe end
    return probe.Distance
end

local function chooseSide(rootCF, lateralWorld, offsetAlongLateral, now)
    local preferred
    if mAbs(offsetAlongLateral) > DodgeConfig.CenterDeadzone then preferred = offsetAlongLateral >= 0 and -1 or 1 else preferred = sideSign end
    if now - sideChosenAt < DodgeConfig.SideCommitTime then return sideSign end
    local origin = rootCF.Position
    local room = clearanceAlong(origin, lateralWorld * preferred)
    if room < DodgeConfig.ClearanceMin then
        local otherRoom = clearanceAlong(origin, lateralWorld * -preferred)
        if otherRoom > room then preferred = -preferred end
    end
    if preferred ~= sideSign then sideSign = preferred; sideChosenAt = now end
    return sideSign
end

local function boxExtentAlong(axis)
    return DodgeConfig.BoxWidth * 0.5 * mAbs(axis.X) + DodgeConfig.BoxHeight * 0.5 * mAbs(axis.Y) + DodgeConfig.BoxDepth * 0.5 * mAbs(axis.Z)
end

local function staticPushout(lp, r)
    local hx = DodgeConfig.BoxWidth * 0.5 + r + DodgeConfig.ExtraMargin
    local hy = DodgeConfig.BoxHeight * 0.5 + r + DodgeConfig.ExtraMargin
    local hz = DodgeConfig.BoxDepth * 0.5 + r + DodgeConfig.ExtraMargin
    local oy = lp.Y - DodgeConfig.BoxOffsetY
    local px, py, pz = hx - mAbs(lp.X), hy - mAbs(oy), hz - mAbs(lp.Z)
    if px <= 0 or py <= 0 or pz <= 0 then return nil end
    local ax, dp = "X", px
    if pz < dp then ax, dp = "Z", pz end
    if DodgeConfig.Vertical and py * (1 / DodgeConfig.VerticalDamp) < dp then ax, dp = "Y", py end
    if ax == "X" then return lp.X >= 0 and v3new(-1, 0, 0) or v3new(1, 0, 0), dp
    elseif ax == "Z" then return lp.Z >= 0 and v3new(0, 0, -1) or v3new(0, 0, 1), dp end
    return oy >= 0 and v3new(0, -1, 0) or v3new(0, 1, 0), dp
end

local function safeOffset(origin, dir, dist)
    local res = Workspace:Raycast(origin, dir * (dist + DodgeConfig.WallPadding), dodgeRay)
    if not res then return dist end
    local allowed = res.Distance - DodgeConfig.WallPadding
    if allowed <= 0.05 then return 0 end
    return mMin(dist, allowed)
end

local function stripDrag(push)
    if not DodgeConfig.NoDrag then return push end
    for i = 1, #BlockedAxes do local ax = BlockedAxes[i]; local al = push:Dot(ax); if al > 0 then push = push - ax * al end end
    return push
end

local function applyPush(push, urgent, now)
    push = stripDrag(push); local len = push.Magnitude
    if len < DodgeConfig.MinStep then return false end
    local dir = push.Unit
    if urgent then len = mMin(len * DodgeConfig.UrgentSmoothing, DodgeConfig.UrgentStep)
    else len = mMin(len * DodgeConfig.Smoothing, DodgeConfig.MaxStep) end
    if len < DodgeConfig.MinStep then return false end
    local origin = DodgeRoot.Position; local allowed = safeOffset(origin, dir, len)
    if allowed <= 0.02 then
        local side = v3new(-dir.Z, 0, dir.X)
        if side.Magnitude < 0.05 then return false end
        side = stripDrag(side.Unit)
        if side.Magnitude < 0.05 then return false end
        side = side.Unit; local sideAllowed = safeOffset(origin, side, len)
        if sideAllowed <= 0.02 then
            local flipped = stripDrag(-side)
            if flipped.Magnitude < 0.05 then return false end
            side = flipped.Unit; sideAllowed = safeOffset(origin, side, len)
        end
        if sideAllowed <= 0.02 then return false end
        sideSign = -sideSign; sideChosenAt = now; dir = side; allowed = sideAllowed
    end
    local offset = dir * allowed
    if mAbs(offset.Y) < 0.05 then
        local ground = Workspace:Raycast(DodgeRoot.Position + offset + v3new(0, 5, 0), v3new(0, -DodgeConfig.GroundDrop, 0), dodgeRay)
        if not ground then return false end
    end
    local vel = DodgeRoot.AssemblyLinearVelocity
    DodgeRoot.CFrame = DodgeRoot.CFrame + offset
    if DodgeConfig.KeepVelocity then DodgeRoot.AssemblyLinearVelocity = vel end
    return true
end

local function resolveOnce(now, dt)
    if not DodgeRoot or not DodgeRoot.Parent or not DodgeHum or DodgeHum.Health <= 0 then return false end
    local rootCF = DodgeRoot.CFrame; local origin = rootCF.Position
    local ownVel = DodgeRoot.AssemblyLinearVelocity
    local sweepHorizon = dt + (DodgeConfig.Predictive and DodgeConfig.PredictTime or 0)
    local leadHorizon = sweepHorizon + DodgeConfig.LeadTime
    local push, urgent, count = v3zero, false, 0
    for i = #BlockedAxes, 1, -1 do BlockedAxes[i] = nil end
    for _, part in ipairs(BallCache) do
        if part and part.Parent then
            local pos = part.Position
            if (pos - origin).Magnitude < DodgeConfig.ScanRadius then
                local r = part.Size.X * 0.5; local vel = velocityOf(part, pos, now)
                local rel = vel - ownVel; local speed = rel.Magnitude
                local travel = rel * leadHorizon; local steps = 1; local span = travel.Magnitude
                if span > DodgeConfig.SubstepStud then steps = mClamp(mCeil(span / DodgeConfig.SubstepStud), 1, DodgeConfig.MaxSubsteps) end
                local threatened, threatTime = false, mHuge
                for si = 0, steps do
                    local frac = si / steps; local sp = pos + travel * frac; local lp = rootCF:PointToObjectSpace(sp)
                    if boxGap(lp, r) < DodgeConfig.ExtraMargin then threatened = true; threatTime = frac * leadHorizon; break end
                end
                if threatened then
                    local escape = nil
                    if speed > DodgeConfig.SlowSpeed then
                        local approach = rel.Unit; tInsert(BlockedAxes, approach)
                        local lateral = v3new(approach.Z, 0, -approach.X)
                        if lateral.Magnitude > 0.05 then
                            lateral = lateral.Unit; local offVec = pos - origin; local sep = offVec:Dot(lateral)
                            local sign = chooseSide(rootCF, lateral, sep, now)
                            local localLat = rootCF:VectorToObjectSpace(lateral)
                            local corridor = boxExtentAlong(localLat) + r + DodgeConfig.ExtraMargin + DodgeConfig.ClearMargin
                            local needed = corridor - mAbs(sep)
                            if sep * sign > 0 then needed = corridor + mAbs(sep) end
                            if needed > 0 then escape = lateral * sign * needed end
                        end
                    end
                    if not escape then
                        local lp = rootCF:PointToObjectSpace(pos); local direction, depth = staticPushout(lp, r)
                        if direction then escape = rootCF:VectorToWorldSpace(direction) * depth end
                    end
                    if escape then
                        if threatTime <= DodgeConfig.UrgentTime then urgent = true end
                        push = push + escape; count = count + 1
                    end
                end
            end
        end
    end
    if count == 0 then return false end
    lastEngaged = now; push = v3new(push.X, push.Y * DodgeConfig.VerticalDamp, push.Z)
    return applyPush(push, urgent, now)
end

local function gojoStep(dt)
    if not States.AutoDodge then return end

    if not DodgeRoot or not DodgeRoot.Parent or not DodgeHum or DodgeHum.Health <= 0 then return end
    if now - lastEngaged > 1 then sideChosenAt = 0 end
    for _ = 1, 3 do local ok, res = pcall(resolveOnce, now, dt); if not ok or not res then break end end
end

local AutoPlayConn = nil
local AutoPlayThread = nil
local AutoPlayEmergencyDest = nil
local AutoPlayRunning = false
local AutoPlayMapCenter = Vector3.new(0, 10, 0)
local AutoPlaySafeRadius = 60
local AutoPlayStuckTimeout = 8
local AutoPlayArriveDist = 4
local function AutoPlay_GetNil(name, class)
    if not getnilinstances then return nil end
    for _, v in next, getnilinstances() do if v.ClassName == class and v.Name == name then return v end end
    return nil
end
local function AutoPlay_GetTargetBalls()
    local targets = {}
    local balls = Workspace:FindFirstChild("Balls")
    if balls then
        for _, name in ipairs({"Big Ball","Speed Ball","Death Ball","Duo Ball","Destruction Ball","Invis Ball"}) do
            local b = balls:FindFirstChild(name)
            if b and b:IsA("BasePart") then table.insert(targets, b) end
        end
        local ch = balls:GetChildren()
        for i = 2, 4 do if ch[i] and ch[i]:IsA("BasePart") then table.insert(targets, ch[i]) end end
    end
    for _, name in ipairs({"Virus Ball","Invis Ball","Destruction Ball","Speed Ball"}) do
        local b = AutoPlay_GetNil(name, "Part")
        if b then table.insert(targets, b) end
    end
    return targets
end
local function AutoPlay_CheckClear(pos, dir, dist)
    local r = Workspace:Raycast(pos + Vector3.new(0, 2, 0), dir * dist)
    return r and r.Distance or dist
end
local function AutoPlay_GetEscape(threatPos, threatVel, playerPos)
    local toPlayer = playerPos - threatPos; local threatDir = threatVel.Unit; local proj = toPlayer:Dot(threatDir)
    if proj < 0 then return nil end
    local closest = threatPos + threatDir * proj; local distToPath = (playerPos - closest).Magnitude
    if distToPath > 8 then return nil end
    local timeToHit = proj / threatVel.Magnitude
    if timeToHit > 1.5 or timeToHit < 0 then return nil end
    local right = Vector3.new(-threatDir.Z, 0, threatDir.X).Unit; local dirs = {right, -right, -threatDir, threatDir}; local bestDir, bestClear = nil, 0
    for _, d in ipairs(dirs) do
        local clear = AutoPlay_CheckClear(playerPos, d, 25)
        if clear > bestClear then bestClear = clear; bestDir = d end
    end
    return bestDir and (playerPos + bestDir * 12) or nil
end
local function AutoPlay_IsGround(pos) return Workspace:Raycast(pos + Vector3.new(0, 5, 0), Vector3.new(0, -25, 0)) ~= nil end
local function AutoPlay_RandomPointNear(origin)
    for _ = 1, 6 do
        local angle = math.random() * math.pi * 2; local distance = math.random(10, AutoPlaySafeRadius)
        local p = origin + Vector3.new(math.cos(angle) * distance, 0, math.sin(angle) * distance)
        if AutoPlay_IsGround(p) then return p end
    end
    return AutoPlayMapCenter + Vector3.new(0, 10, 0)
end
local function AutoPlay_WalkTo(humanoid, rootPart, destination)
    local path = PathfindingService:CreatePath({AgentRadius = 2, AgentHeight = 5, AgentCanJump = true})
    local success = pcall(function() path:ComputeAsync(rootPart.Position, destination) end)
    if not success or path.Status ~= Enum.PathStatus.Success then
        humanoid:MoveTo(destination)
        local finished = false; local conn = humanoid.MoveToFinished:Connect(function() finished = true end); local t0 = tick()
        while not finished and AutoPlayRunning do
            if tick() - t0 > AutoPlayStuckTimeout or AutoPlayEmergencyDest then break end
            task.wait(0.2)
        end
        conn:Disconnect()
    else
        for _, waypoint in ipairs(path:GetWaypoints()) do
            if not AutoPlayRunning or AutoPlayEmergencyDest then return end
            if waypoint.Action == Enum.PathWaypointAction.Jump then humanoid.Jump = true end
            humanoid:MoveTo(waypoint.Position)
            local finished = false; local conn = humanoid.MoveToFinished:Connect(function() finished = true end); local t0 = tick()
            while not finished and AutoPlayRunning do
                if tick() - t0 > AutoPlayStuckTimeout or AutoPlayEmergencyDest then break end
                if (rootPart.Position - waypoint.Position).Magnitude <= AutoPlayArriveDist then break end
                task.wait(0.1)
            end
            conn:Disconnect()
        end
    end
end
local function AutoPlay_Heartbeat()
    if not AutoPlayRunning then return end
    local char = LP.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum or hum.Health <= 0 then return end
    local rootPos = root.Position; local fall = Workspace:Raycast(rootPos + Vector3.new(0, 3, 0), Vector3.new(0, -60, 0))
    if not fall or fall.Distance > 35 then
        root.CFrame = CFrame.new(AutoPlayMapCenter + Vector3.new(0, 10, 0))
        AutoPlayEmergencyDest = AutoPlay_RandomPointNear(root.Position); hum:MoveTo(AutoPlayEmergencyDest)
        return
    end
    local bestEscape, minDist = nil, math.huge
    for _, part in ipairs(AutoPlay_GetTargetBalls()) do
        if part and (part.Parent or AutoPlay_GetNil(part.Name, "Part")) then
            local vel = part.AssemblyLinearVelocity
            if vel.Magnitude > 2 then
                local esc = AutoPlay_GetEscape(part.Position, vel, rootPos)
                if esc then
                    local d = (part.Position - rootPos).Magnitude
                    if d < minDist then minDist = d; bestEscape = esc end
                end
            end
        end
    end
    if bestEscape then AutoPlayEmergencyDest = bestEscape; hum:MoveTo(bestEscape) end
end

-- VERSÃO BLINDADA DO LOOP (SEM CONTINUE)
local function AutoPlay_WanderLoop()
    while AutoPlayRunning do
        local char = LP.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                if AutoPlayEmergencyDest then
                    local dest = AutoPlayEmergencyDest
                    AutoPlayEmergencyDest = nil
                    hum:MoveTo(dest)
                    local finished = false
                    local conn = hum.MoveToFinished:Connect(function() finished = true end)
                    local t0 = tick()
                    while not finished and AutoPlayRunning do
                        if tick() - t0 > AutoPlayStuckTimeout or AutoPlayEmergencyDest then break end
                        task.wait(0.15)
                    end
                    conn:Disconnect()
                else
                    pcall(function() AutoPlay_WalkTo(hum, root, AutoPlay_RandomPointNear(root.Position)) end)
                end
            else
                task.wait(1)
            end
        else
            task.wait(1)
        end
    end
end

local function AutoPlay_Toggle(on)
    if not on then
        AutoPlayRunning = false
        if AutoPlayConn then pcall(function() AutoPlayConn:Disconnect() end); AutoPlayConn = nil end
        AutoPlayThread = nil
        local char = LP.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum:MoveTo(Vector3.zero) end
        end
        return
    end
    local char = LP.Character; if not char then States.AutoPlay = false; return end
    local hum = char:FindFirstChildOfClass("Humanoid"); local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then States.AutoPlay = false; return end
    AutoPlayRunning = true
    AutoPlayConn = RunService.Heartbeat:Connect(function() pcall(AutoPlay_Heartbeat) end)
    AutoPlayThread = task.spawn(AutoPlay_WanderLoop)
end

local function ripple(card, input)
    local sc = fitScale.Scale; local abs = card.AbsolutePosition; local size = card.AbsoluteSize
    local x, y = size.X / sc / 2, size.Y / sc / 2
    if input and input.Position then x = (input.Position.X - abs.X) / sc; y = (input.Position.Y - abs.Y) / sc end
    local r = new("Frame", { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(0, 0), BackgroundColor3 = WHITE, BackgroundTransparency = 0.82, BorderSizePixel = 0, ZIndex = 2, Parent = card })

local function tabY(i) return TAB_TOP + (i - 1) * (TAB_H + TAB_GAP) end
local function renderTab(name, animate)
    local data = tabButtons[name]; if not data then return end
    local c = Theme.c; local d = animate and 0.22 or 0; local active = activeTab == name; local col = active and c.textActive or (data.hover and TEXT or TEXT_DIM)
    apply(data.icon, { ImageColor3 = col, Size = active and UDim2.fromOffset(20, 20) or UDim2.fromOffset(18, 18), Position = active and UDim2.new(0, 11, 0.5, -10) or UDim2.new(0, 12, 0.5, -9) }, d, Enum.EasingStyle.Back)
    apply(data.label, { TextColor3 = col }, d)
end
local function playEntrance(page)
    entranceToken = entranceToken + 1; local my = entranceToken
    contentGroup.GroupTransparency = 0.65; tween(contentGroup, { GroupTransparency = 0 }, 0.28, Enum.EasingStyle.Quad)
    local wraps = {}
    for _, ch in ipairs(page:GetChildren()) do if ch:IsA("Frame") and ch:GetAttribute("Entrance") then wraps[#wraps + 1] = ch end end
    table.sort(wraps, function(a, b) return a.LayoutOrder < b.LayoutOrder end)
    for i, w in ipairs(wraps) do
        local card = w:FindFirstChild("Card")
        if card then
            card.Position = UDim2.new(0, 18, 0, 0)
            task.delay((i - 1) * 0.04, function() if entranceToken == my and card.Parent then tween(card, { Position = UDim2.new(0, 0, 0, 0) }, 0.4, Enum.EasingStyle.Quint) end end)
        end
    end
end
local function selectTab(name, silent, force)
    if activeTab == name and not force then return end
    local prev = activeTab; activeTab = name
    if not silent then playSound(SOUND_TAB, 0.35) end
    tween(tabIndicator, { Position = UDim2.fromOffset(8, tabY(tabButtons[name].index)) }, silent and 0 or 0.32, Enum.EasingStyle.Quint)
    if prev then renderTab(prev, true) end
    renderTab(name, true)
    for n, p in pairs(pages) do p.Visible = (n == name) end
    pages[name].CanvasPosition = Vector2.new(0, 0)
    playEntrance(pages[name])
end
for i, tab in ipairs(TABS) do
    local btn = new("TextButton", { Size = UDim2.fromOffset(tabW, TAB_H), Position = UDim2.fromOffset(8, tabY(i)), BackgroundTransparency = 1, Text = "", AutoButtonColor = false, ZIndex = 3, Parent = sidebar })
    local icon = new("ImageLabel", { Size = UDim2.fromOffset(18, 18), Position = UDim2.new(0, 12, 0.5, -9), BackgroundTransparency = 1, Image = tab.icon, ImageColor3 = TEXT_DIM, ZIndex = 4, Parent = btn })
    local label = newLabel(btn, tab.name, 12, FONT_MAIN, TEXT_DIM, { Size = UDim2.new(1, -44, 1, 0), Position = UDim2.new(0, 40, 0, 0), ZIndex = 4 })
    local data = { btn = btn, icon = icon, label = label, index = i, hover = false }
    tabButtons[tab.name] = data
    btn.MouseEnter:Connect(function() data.hover = true renderTab(tab.name, true) end)
    btn.MouseLeave:Connect(function() data.hover = false renderTab(tab.name, true) end)
    btn.MouseButton1Click:Connect(function() selectTab(tab.name) end)
end
Theme.reg(sidebar, function(_, a) for n in pairs(tabButtons) do renderTab(n, a) end end)

local posX, posY, tgtX, tgtY = 0, 0, 0, 0
local dragging, dragOrigin, startX, startY = false, nil, 0, 0
local isMin, isAnim, resizing = false, false, false
local function clampPos(x, y)
    local vp = viewport(); local sc = fitScale.Scale; local halfW = winW * sc / 2; local halfH = winH * sc / 2
    return math.clamp(x, -(vp.X / 2 - halfW + 20), vp.X / 2 - halfW + 20), math.clamp(y, -(vp.Y / 2 - halfH + 20), vp.Y / 2 - halfH + 20)
end

grip.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
    if isAnim or isMin or resizing then return end
    local isTouch = input.UserInputType == Enum.UserInputType.Touch; local startP = isTouch and input.Position or UserInputService:GetMouseLocation()
    local vp = viewport(); local startWinW, startWinH = winW, winH; local startScale = fitScale.Scale; local startPosX, startPosY = posX, posY
    local topLeftX = vp.X / 2 + startPosX - startWinW * startScale / 2; local topLeftY = vp.Y / 2 + startPosY - startWinH * startScale / 2
    resizing = true
    local c1, c2
    local function finish() resizing = false; if c1 then c1:Disconnect() end; if c2 then c2:Disconnect() end end
    c1 = UserInputService.InputChanged:Connect(function(i)
        local ok = (isTouch and i.UserInputType == Enum.UserInputType.Touch) or (not isTouch and i.UserInputType == Enum.UserInputType.MouseMovement)
        if not ok then return end
        local p = isTouch and i.Position or UserInputService:GetMouseLocation(); local d = p - startP
        local targetAbsW = math.clamp(startWinW * startScale + d.X, MIN_W * startScale, MAX_W * startScale)
        local targetAbsH = math.clamp(startWinH * startScale + d.Y, MIN_H * startScale, MAX_H * startScale)
        local curScale = fitScale.Scale
        local newWinW = math.clamp(targetAbsW / curScale, MIN_W, MAX_W); local newWinH = math.clamp(targetAbsH / curScale, MIN_H, MAX_H)
        winW, winH = newWinW, newWinH; root.Size = UDim2.fromOffset(newWinW, newWinH); updateFit()
        local finalScale = fitScale.Scale; local newCX = topLeftX + (winW * finalScale) / 2; local newCY = topLeftY + (winH * finalScale) / 2
        posX, posY = newCX - vp.X / 2, newCY - vp.Y / 2; tgtX, tgtY = posX, posY
        root.Position = UDim2.new(0.5, posX, 0.5, posY)
    end)
    c2 = UserInputService.InputEnded:Connect(function(i)
        local ok = (isTouch and i.UserInputType == Enum.UserInputType.Touch) or (not isTouch and i.UserInputType == Enum.UserInputType.MouseButton1)
        if ok then finish() end
    end)
end)

    if isAnim or isMin or resizing then return end

    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        local d = input.Position - dragOrigin; tgtX, tgtY = clampPos(startX + d.X, startY + d.Y)
    end
end)
bind(UserInputService.InputEnded, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
end)

        local k = math.min(1, dt * 22)

        if math.abs(tgtX - posX) < 0.25 then posX = tgtX end

        if not (isAnim or isMin) then root.Position = UDim2.new(0.5, posX, 0.5, posY) end
    end
    premiumClock = premiumClock + dt; premiumGrad.Offset = Vector2.new(-1 + ((premiumClock * 0.45) % 2.4) - 0.2, 0)
    if Options.rainbow_theme then
        if not _G._rainbowClock then _G._rainbowClock = 0 end
        _G._rainbowClock = _G._rainbowClock + dt
        if _G._rainbowClock > 0.1 then _G._rainbowClock = 0; Theme.set(Color3.fromHSV((os.clock() * 0.15) % 1, 0.7, 0.9)) end
    end
end)

local ballHolder, ballPos = nil, Vector2.new(16, 16)
local function clampBall(p) local vp = viewport(); return Vector2.new(math.clamp(p.X, 6, vp.X - BALL - 6), math.clamp(p.Y, 6, vp.Y - BALL - 6)) end
local restore
local function createBall()
    ballPos = clampBall(ballPos)
    ballHolder = new("Frame", { Name = "BallHolder", Size = UDim2.fromOffset(BALL, BALL), Position = UDim2.fromOffset(ballPos.X, ballPos.Y), BackgroundTransparency = 1, ZIndex = 300, Parent = gui })
    local hScale = new("UIScale", { Scale = 0, Parent = ballHolder })
    local ball = new("TextButton", { Size = UDim2.fromOffset(BALL, BALL), BackgroundColor3 = BLACK, Text = "", BorderSizePixel = 0, AutoButtonColor = false, ClipsDescendants = true, ZIndex = 300, Parent = ballHolder })
    corner(ball, 16); local ballStroke = stroke(ball, Theme.c.accent, 1.5, 0.4)
    new("ImageLabel", { Size = UDim2.fromOffset(40, 40), Position = UDim2.new(0.5, -20, 0.5, -20), BackgroundTransparency = 1, Image = LOGO_BALL_ICON, ImageColor3 = WHITE, ScaleType = Enum.ScaleType.Fit, ZIndex = 305, Parent = ball })
    if HW.EMOJI then new("TextLabel", { AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -1, 0, 1), Size = UDim2.fromOffset(20, 20), BackgroundTransparency = 1, Font = FONT_MAIN, Text = HW.PUMPKIN, TextSize = 14, TextColor3 = WHITE, ZIndex = 306, Parent = ball }) end
    tween(hScale, { Scale = 1 }, 0.45, Enum.EasingStyle.Back)
    ball.MouseEnter:Connect(function() tween(ball, { BackgroundColor3 = Color3.fromRGB(34, 18, 46) }, 0.15); tween(ballStroke, { Transparency = 0 }, 0.15); tween(hScale, { Scale = 1.07 }, 0.15, Enum.EasingStyle.Back) end)
    ball.MouseLeave:Connect(function() tween(ball, { BackgroundColor3 = BLACK }, 0.15); tween(ballStroke, { Transparency = 0.4 }, 0.15); tween(hScale, { Scale = 1 }, 0.15) end)
    local bDrag, moved, bStart, bOrigin = false, false, nil, nil
    ball.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then bDrag, moved = true, false; bStart = input.Position; bOrigin = ballPos end
    end)
    bind(UserInputService.InputChanged, function(input)
        if bDrag and ballHolder and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - bStart
            if math.abs(d.X) > 8 or math.abs(d.Y) > 8 then moved = true end
            if moved then ballPos = clampBall(Vector2.new(bOrigin.X + d.X, bOrigin.Y + d.Y)); ballHolder.Position = UDim2.fromOffset(ballPos.X, ballPos.Y) end
        end
    end)
    bind(UserInputService.InputEnded, function(input)
        if bDrag and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            bDrag = false
            if moved and ballHolder then
                local vp = viewport(); local snapX = (ballPos.X + BALL / 2 < vp.X / 2) and 10 or (vp.X - BALL - 10)
                ballPos = clampBall(Vector2.new(snapX, ballPos.Y)); tween(ballHolder, { Position = UDim2.fromOffset(ballPos.X, ballPos.Y) }, 0.3, Enum.EasingStyle.Back)
            elseif not moved then restore() end
        end
    end)
end
local function minimize()
    if isAnim or isMin then return end
    isAnim = true; local cx, cy = ballPos.X + BALL / 2, ballPos.Y + BALL / 2
    tween(canvas, { GroupTransparency = 1 }, 0.24); if HW.decor then tween(HW.decor, { GroupTransparency = 1 }, 0.24) end
    tween(popScale, { Scale = 0.35 }, 0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    tween(root, { Position = UDim2.fromOffset(cx, cy) }, 0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    setShadow(false, 0.2); task.wait(0.3); root.Visible = false; createBall(); isMin, isAnim = true, false
end
restore = function()
    if isAnim or not isMin then return end
    isAnim = true; local lx, ly = ballPos.X + BALL / 2, ballPos.Y + BALL / 2; local holder = ballHolder; ballHolder = nil
    if holder then
        local sc = holder:FindFirstChildOfClass("UIScale")
        if sc then tween(sc, { Scale = 0 }, 0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.In) end
        Debris:AddItem(holder, 0.22)
    end
    task.wait(0.18)
    root.Visible = true; root.Position = UDim2.fromOffset(lx, ly); canvas.GroupTransparency = 1; popScale.Scale = 0.35
    tween(root, { Position = UDim2.new(0.5, posX, 0.5, posY) }, 0.45, Enum.EasingStyle.Back)
    tween(popScale, { Scale = 1 }, 0.45, Enum.EasingStyle.Back)
    tween(canvas, { GroupTransparency = 0 }, 0.3)
    if HW.decor then tween(HW.decor, { GroupTransparency = HW.DECOR_ALPHA }, 0.5) end
    setShadow(true, 0.4)
    task.delay(0.46, function() isMin, isAnim = false, false end)
end
